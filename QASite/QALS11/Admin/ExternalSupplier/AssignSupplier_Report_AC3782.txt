    function assign() {
        var frm = $("div#frma form")
        var tbl = $("table#AssignSupplier", frm)
        if ($("tr:gt(0) input:checkbox:checked[name^='AssignSupplier']", tbl).length == 0) {
            $("#dialog").ebomodal({ modalType: "Close", modalText: "Please select at least one Supplier to assign" });
            return;
        }
        var isValid = true
        
        $("tr:gt(0) input:checkbox[name^='AssignSupplier']", tbl).each(function() {
            if ($(this).parent()[0].tagName === "TH") return true;
            var oC = $(this);
            var tr = oC.closest("td").closest("tr");
            var txtCode = $("td input[id*='TextboxBusSupplierCode']", tr);
            var objImg = $("img", txtCode.closest("td"));
            var code = jQuery.trim(txtCode.val());
            
            if($(this).is(":checked")) {
                if(code == ""){
                    txtCode.addClass("input-validation-error")
                    objImg.attr("src", objImg.attr("esrc")); 
                    objImg.addClass("TipImage"); 
                    objImg.ebotooltip();
                    isValid = false;
                }
                else{
                    objImg.attr("src", objImg.attr("dsrc")); 
                    objImg.removeClass("TipImage");
                    txtCode.removeClass("input-validation-error")
                }
            }
            else{
                objImg.attr("src", objImg.attr("dsrc")); 
                txtCode.removeClass("input-validation-error")
            }
        });
        if(!isValid) 
        {
            $("div#pgerr").show()
            return; 
        }        
        frm.submit()
    }
